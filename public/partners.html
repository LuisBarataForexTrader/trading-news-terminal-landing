Live Demo

One snippet. Instant market intelligence.

Paste two lines of code on any page and your users get a live, auto-updating financial news feed — no backend required on your end.

partner-site.com
Embed code
<!-- Add to your site -->
<div id="tnt-widget"></div>
<script
  src="https://tradingnewsterminal.com/widget.js"
  data-key="YOUR_PARTNER_KEY"
  data-theme="dark"
  data-topics="forex,commodities"
  data-impact="high,medium"
></script>
import TNTWidget from '@tnt/widget';

const widget = new TNTWidget({
  key: 'YOUR_PARTNER_KEY',
  container: '#tnt-widget',
  theme: 'dark',
  topics: ['forex', 'commodities'],
  impact: ['high', 'medium'],
  onNews: (item) => console.log(item)
});

widget.init();
